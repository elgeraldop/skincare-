    <script>
        // Navigation
        function showSection(id) {
            document.querySelectorAll('.section').forEach(s => s.classList.remove('active'));
            document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
            document.getElementById(id).classList.add('active');
            event.target.classList.add('active');
        }
        
        // Checkboxes
        function toggleCheck(el) {
            el.classList.toggle('checked');
            saveState();
        }
        
        function toggleChecklist(el) {
            el.classList.toggle('checked');
            el.nextElementSibling.classList.toggle('checked');
            updateStreak();
            saveState();
        }
        
        // Week selector
        function setWeek(week) {
            document.querySelectorAll('.week-btn').forEach(b => b.classList.remove('active'));
            event.target.classList.add('active');
            
            const fill = document.getElementById('progressFill');
            const text = document.getElementById('progressText');
            const weekLabel = document.getElementById('progressWeek');
            
            const phases = {
                1: { pct: 15, text: 'Fase inicial: limpiador + sérum + hidratante + SPF', label: 'Semana 1-2' },
                3: { pct: 35, text: 'Fase activa: introduces retinol 2-3 noches/semana', label: 'Semana 3+' },
                6: { pct: 60, text: 'Fase consolidación: ajustas frecuencia de retinol', label: 'Semana 6+' },
                12: { pct: 100, text: 'Fase mantenimiento: resultados visibles establecidos', label: '3 meses' }
            };
            
            fill.style.width = phases[week].pct + '%';
            text.textContent = phases[week].text;
            weekLabel.textContent = phases[week].label;
        }
        
        // Streak counter
        function updateStreak() {
            const checked = document.querySelectorAll('.checklist-box.checked').length;
            document.getElementById('streakCount').textContent = checked;
        }
        
        // Reset
        function resetAll() {
            if(confirm('¿Reiniciar todo el progreso de hoy?')) {
                document.querySelectorAll('.checkbox, .checklist-box').forEach(el => {
                    el.classList.remove('checked');
                });
                document.querySelectorAll('.checklist-text').forEach(el => {
                    el.classList.remove('checked');
                });
                updateStreak();
                saveState();
            }
        }
        
        // Share/Save
        function shareProgress() {
            const data = {
                date: new Date().toLocaleDateString('es-VE'),
                streak: document.getElementById('streakCount').textContent,
                checks: document.querySelectorAll('.checkbox.checked, .checklist-box.checked').length
            };
            
            if (navigator.share) {
                navigator.share({
                    title: 'Mi rutina skincare',
                    text: `Día ${data.streak}/6 completado · ${data.checks} pasos hechos`
                });
            } else {
                alert('Progreso guardado localmente en tu iPhone');
            }
        }
        
        // Save/Load state
        function saveState() {
            const state = {
                checks: Array.from(document.querySelectorAll('.checkbox.checked, .checklist-box.checked')).map(el => {
                    const items = Array.from(document.querySelectorAll('.checkbox, .checklist-box'));
                    return items.indexOf(el);
                })
            };
            localStorage.setItem('skincare_geraldo', JSON.stringify(state));
        }
        
        function loadState() {
            try {
                const saved = localStorage.getItem('skincare_geraldo');
                if (saved) {
                    const state = JSON.parse(saved);
                    const items = document.querySelectorAll('.checkbox, .checklist-box');
                    state.checks.forEach(idx => {
                        if (items[idx]) items[idx].classList.add('checked');
                    });
                    updateStreak();
                }
            } catch(e) {}
        }
        
        // Init
        loadState();
        updateStreak();
        
        // Auto-set week based on first visit
        if (!localStorage.getItem('skincare_start')) {
            localStorage.setItem('skincare_start', Date.now());
        }
        
        const daysSince = Math.floor((Date.now() - parseInt(localStorage.getItem('skincare_start'))) / 86400000);
        const currentWeek = Math.min(12, Math.max(1, Math.floor(daysSince / 7) + 1));
        
        // PWA Service Worker
        if ('serviceWorker' in navigator) {
            navigator.serviceWorker.register('data:text/javascript,').catch(() => {});
        }
    </script>
